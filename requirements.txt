# Tier-2 OCR, for a CV with no text layer (cv_ocr.py). Only reached when the
# normal extraction comes back with nothing readable, so a checkout without
# these still runs and still handles every text-based CV.
#
# PyMuPDF renders the page pixels, RapidOCR reads them (PaddleOCR's models
# compiled to ONNX), onnxruntime is the CPU runtime. Pure CPU, no torch, no
# system binaries - a scan reads in about 4-5s a page.
#
# PyMuPDF is AGPL-3.0 / dual-licensed, which is fine for a locally run tool. If
# this ever ships as a hosted service, swap the renderer for pypdfium2 (BSD-3) -
# only cv_ocr._read_pdf touches it.
pymupdf==1.28.2
rapidocr==3.9.2
onnxruntime==1.30.0
